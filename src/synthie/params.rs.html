10/// Oscillator waveform shape.
11#[derive(Debug, Clone, Copy, PartialEq, Eq)]
12#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
13pub enum Waveform {
14    /// Classic square/pulse wave; width controlled by `pulse_width`.
15    Pulse,
16    /// Band-limited sawtooth.
17    Sawtooth,
18    /// Triangle wave.
19    Triangle,
20    /// LFSR-based noise clocked at the oscillator period.
21    Noise,
22    /// 50/50 mix of pulse and sawtooth for a thicker timbre.
23    PulseSaw,
24    /// Pure sine wave.
25    Sine,
26}
27
28impl Waveform {
29    /// Ordered slice of all variants, used for cycling.
30    pub const ALL: &'static [Waveform] = &[
31        Waveform::Pulse,
32        Waveform::Sawtooth,
33        Waveform::Triangle,
34        Waveform::Noise,
35        Waveform::PulseSaw,
36        Waveform::Sine,
37    ];
38
39    /// Short display name shown in the UI.
40    #[must_use]
41    pub fn name(self) -> &'static str {
42        match self {
43            Waveform::Pulse => "Pulse",
44            Waveform::Sawtooth => "Saw",
45            Waveform::Triangle => "Tri",
46            Waveform::Noise => "Noise",
47            Waveform::PulseSaw => "Pls+Saw",
48            Waveform::Sine => "Sine",
49        }
50    }
51
52    /// Return the next variant, wrapping around.
53    #[must_use]
54    pub fn next(self) -> Self {
55        let idx = Self::ALL.iter().position(|&w| w == self).unwrap_or(0);
56        Self::ALL[(idx + 1) % Self::ALL.len()]
57    }
58
59    /// Return the previous variant, wrapping around.
60    #[must_use]
61    pub fn prev(self) -> Self {
62        let idx = Self::ALL.iter().position(|&w| w == self).unwrap_or(0);
63        let len = Self::ALL.len();
64        Self::ALL[(idx + len - 1) % len]
65    }
66}

68/// Arpeggiator playback mode.
69#[cfg(feature = "arp")]
70#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
71#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
72pub enum ArpMode {
73    /// Play notes in ascending order, then wrap to the lowest.
74    #[default]
75    Up,
76    /// Play notes in descending order, then wrap to the highest.
77    Down,
78    /// Ascend to the highest note, then descend to the lowest (no endpoint repeat).
79    UpDown,
80    /// Advance using a Galois LFSR, producing a pseudo-random sequence with a long period.
81    Random,
82}

84/// Arpeggiator section parameters.
85#[cfg(feature = "arp")]
86#[derive(Debug, Clone, PartialEq)]
87#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
88pub struct ArpParams {
89    /// Whether the arpeggiator is active on this channel.
90    pub enabled: bool,
91    /// Step rate in Hz (e.g. `10.0` = 10 notes per second).
92    pub rate: f32,
93    /// Gate length as a fraction of the step duration, `0.0..=1.0`.
94    /// At `0.0` the note is released immediately; at `1.0` it rings until the next step.
95    pub gate: f32,
96    /// Playback mode.
97    pub mode: ArpMode,
98    /// Fixed-size note list. Only entries `0..count` are active.
99    pub notes: [MidiNote; 4],
100    /// Number of active entries in `notes`, `0..=4`.
101    pub count: u8,
102}
103
104#[cfg(feature = "arp")]
105impl Default for ArpParams {
106    fn default() -> Self {
107        Self {
108            enabled: false,
109            rate: 10.0,
110            gate: 0.8,
111            mode: ArpMode::Up,
112            notes: [MidiNote::MIDDLE_C; 4],
113            count: 0,
114        }
115    }
116}

641/// A typed wrapper around a MIDI note byte.
642///
643/// The inner `u8` field is public for ergonomic construction with numeric
644/// literals (`MidiNote(60)`).  No range check is performed; the MIDI spec
645/// defines valid values as 0..=127, but values up to 255 are accepted.
646/// Use [`MidiNote::new_clamped`] when constructing from untrusted input.
647#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
648#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
649pub struct MidiNote(pub u8);
650
651impl MidiNote {
652    /// Middle C (C4).
653    pub const MIDDLE_C: Self = Self(60);
654    /// A4 (440 Hz reference pitch).
655    pub const A4: Self = Self(69);
656
657    /// Clamp `v` to 0..=127 and wrap in `MidiNote`.
658    #[must_use]
659    pub const fn new_clamped(v: u8) -> Self {
660        Self(if v > 127 { 127 } else { v })
661    }
662
663    /// Raw MIDI byte value.
664    #[must_use]
665    pub const fn as_u8(self) -> u8 {
666        self.0
667    }
668}
669
670impl From<u8> for MidiNote {
671    fn from(v: u8) -> Self {
672        Self(v)
673    }
674}
675
676impl Default for MidiNote {
677    fn default() -> Self {
678        Self::MIDDLE_C
679    }
680}

682/// Index of an independent synthesis channel (voice pool + parameter set).
683///
684/// Channel 0 is the default; `NoteOn` / `LoadPatch` without a channel argument
685/// implicitly target it.  Values beyond the engine's `NUM_CHANNELS` limit are
686/// silently ignored.
687#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
688#[cfg_attr(feature = "serde", derive(Serialize, Deserialize))]
689pub struct ChannelNo(pub u8);
690
691impl ChannelNo {
692    /// The implicit default channel used by the channel-less event variants.
693    pub const DEFAULT: Self = Self(0);
694
695    /// Convert to `usize` for array indexing.
696    #[must_use]
697    pub const fn as_usize(self) -> usize {

793/// Messages sent from the UI thread to the audio thread over the event channel.
794#[derive(Default, Debug, Clone)]
795pub enum AudioEvent {
796    /// Immediately silence all voices and clear active note routing on all channels.
797    #[default]
798    Panic,
799    /// Begin sustaining a note at the given MIDI note number on channel 0.
800    NoteOn(MidiNote),
801    /// Release the note at the given MIDI note number on channel 0.
802    NoteOff(MidiNote),
803    /// Replace the parameter set for channel 0 with a new snapshot.
804    LoadPatch(Box<SynthParams>),
805    /// Trigger a one-shot synthesized drum hit.
806    Drum(DrumHit),
807    /// Begin sustaining a note on the given channel at the given MIDI note number.
808    NoteOnChannel(ChannelNo, MidiNote),
809    /// Release the note on the given channel at the given MIDI note number.
810    NoteOffChannel(ChannelNo, MidiNote),
811    /// Replace the parameter set for the given channel with a new snapshot.
812    LoadPatchChannel(ChannelNo, Box<SynthParams>),
813    /// Load an explicit note list into the arpeggiator on the given channel.
814    /// Only entries `0..count` of `notes` are used; `count` must be `<= 4`.
815    #[cfg(feature = "arp")]
816    ArpSetNotes(ChannelNo, [MidiNote; 4], u8),
817    /// Enable or disable the arpeggiator on the given channel at runtime.
818    #[cfg(feature = "arp")]
819    ArpEnabled(ChannelNo, bool),
820}

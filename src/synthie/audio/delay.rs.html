1//! Single-tap digital delay with feedback and wet/dry mix.
2//!
3//! Ring buffer allocated once at construction. No heap allocation in `process()`.
4
5use crate::params::DelayParams;
6
7/// Maximum supported delay time in milliseconds. Determines ring buffer allocation.
8const MAX_DELAY_MS: f32 = 2000.0;
9
10/// Single-tap feedback delay line.
11pub struct Delay {
12    buf: Vec<f32>,
13    write_idx: usize,
14    sample_rate: f32,
15}
16
17impl Delay {
18    /// Allocate a delay line for the given sample rate.
19    ///
20    /// Buffer length = `ceil(MAX_DELAY_MS` / 1000 * `sample_rate)` samples.
21    #[must_use]
22    pub fn new(sample_rate: f32) -> Self {
23        #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
24        let len = ((MAX_DELAY_MS / 1000.0) * sample_rate).ceil() as usize;
25        Self {
26            buf: vec![0.0; len.max(1)],
27            write_idx: 0,
28            sample_rate,
29        }
30    }
31
32    /// Process one sample.
33    ///
34    /// Returns `input` unchanged when `mix < 1e-4` (fast bypass).
35    pub fn process(&mut self, input: f32, params: &DelayParams) -> f32 {
36        if params.mix < 1e-4 {
37            return input;
38        }
39
40        let buf_len = self.buf.len();
41        #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
42        let delay_samples =
43            ((params.time_ms / 1000.0 * self.sample_rate).round() as usize).clamp(1, buf_len);
44        let feedback = params.feedback.clamp(0.0, 0.999);
45
46        let read_idx = (self.write_idx + buf_len - delay_samples) % buf_len;
47        let wet = self.buf[read_idx];
48
49        self.buf[self.write_idx] = input + wet * feedback;
50        self.write_idx = (self.write_idx + 1) % buf_len;
51
52        input * (1.0 - params.mix) + wet * params.mix
53    }
54}

56#[cfg(test)]
57mod tests {
58    use super::*;
59
60    fn params(time_ms: f32, feedback: f32, mix: f32) -> DelayParams {
61        DelayParams {
62            time_ms,
63            feedback,
64            mix,
65        }
66    }
67
68    #[test]
69    #[allow(clippy::float_cmp)]
70    fn bypass_at_zero_mix() {
71        let mut d = Delay::new(44100.0);
72        let p = params(100.0, 0.5, 0.0);
73        for &x in &[-1.0_f32, -0.5, 0.0, 0.5, 1.0] {
74            assert_eq!(d.process(x, &p), x, "zero-mix bypass failed for {x}");
75        }
76    }
77
78    #[test]
79    fn delayed_signal_appears_at_correct_offset() {
80        let sr = 44100.0_f32;
81        let time_ms = 10.0_f32;
82        #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
83        let delay_samples = (time_ms / 1000.0 * sr).round() as usize; // 441
84
85        let mut d = Delay::new(sr);
86        let p = params(time_ms, 0.0, 1.0);
87
88        // Inject impulse; wet output is 0 (buffer was silent).
89        let out0 = d.process(1.0, &p);
90        assert!(out0.abs() < 1e-4, "expected ~0 immediately, got {out0}");
91
92        // Feed silence for delay_samples-1 more calls.
93        for _ in 0..(delay_samples - 1) {
94            d.process(0.0, &p);
95        }
96
97        // On the delay_samples-th call, the impulse should come back.
98        let out_delayed = d.process(0.0, &p);
99        assert!(
100            (out_delayed - 1.0).abs() < 1e-4,
101            "impulse not found at delay offset: got {out_delayed}"
102        );
103    }
104
105    #[test]
106    fn feedback_energy_decays() {
107        let mut d = Delay::new(44100.0);
108        // 1 ms delay so the echo period is short and we can measure many cycles fast.
109        let p = params(1.0, 0.5, 1.0);
110
111        d.process(1.0, &p); // inject impulse
112
113        let outputs: Vec<f32> = (0..200).map(|_| d.process(0.0, &p)).collect();
114        let first: f32 = outputs[..100].iter().map(|x| x * x).sum();
115        let second: f32 = outputs[100..].iter().map(|x| x * x).sum();
116        assert!(
117            second < first,
118            "feedback should decay: first={first}, second={second}"
119        );
120    }
121
122    #[test]
123    fn no_nan_at_boundary_params() {
124        let mut d = Delay::new(44100.0);
125        for &time_ms in &[1.0_f32, 1000.0, 2000.0] {
126            for &feedback in &[0.0_f32, 0.5, 0.95] {
127                for &mix in &[0.0_f32, 0.5, 1.0] {
128                    let p = params(time_ms, feedback, mix);
129                    for &x in &[-1.0_f32, 0.0, 1.0] {
130                        let out = d.process(x, &p);
131                        assert!(
132                            out.is_finite(),
133                            "NaN/Inf: time={time_ms}, fb={feedback}, mix={mix}, in={x}, out={out}"
134                        );
135                    }
136                }
137            }
138        }
139    }

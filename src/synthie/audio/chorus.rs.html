1//! BBD-style chorus: 3 modulated taps from a shared delay line.
2//!
3//! Three sine LFOs at 0/120/240-degree phase spread modulate the read position
4//! around a 15 ms center delay. Linear interpolation between samples.
5//!
6//! Hot path uses sin/cos rotation (no `sin()` calls per sample) to stay within
7//! real-time audio budget at high LFO rates.
8
9use core::f32::consts::TAU;
10
11use crate::params::ChorusParams;
12
13/// Maximum chorus delay line length in milliseconds.
14const MAX_CHORUS_DELAY_MS: f32 = 50.0;
15
16/// Center delay offset in milliseconds (read position when depth = 0).
17const CENTER_DELAY_MS: f32 = 15.0;
18
19/// Number of modulated taps.
20const NUM_TAPS: usize = 3;
21
22/// BBD-style 3-tap chorus with sine LFO modulation.
23pub struct Chorus {
24    buf: Vec<f32>,
25    write_idx: usize,
26    /// Per-tap LFO state as (sin, cos) pairs — advanced via rotation each sample.
27    lfo: [(f32, f32); NUM_TAPS],
28    /// Cached rotation step recomputed only when `params.rate` changes.
29    sin_inc: f32,
30    cos_inc: f32,
31    cached_rate: f32,
32    sample_rate: f32,
33}
34
35impl Chorus {
36    /// Allocate chorus delay line for the given sample rate.
37    ///
38    /// LFO phases are initialised at 0°, 120°, 240°.
39    #[must_use]
40    pub fn new(sample_rate: f32) -> Self {
41        #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
42        let len = ((MAX_CHORUS_DELAY_MS / 1000.0) * sample_rate).ceil() as usize;
43        let phases = [0.0_f32, TAU / 3.0, 2.0 * TAU / 3.0];
44        let lfo = phases.map(crate::math::sin_cos);
45        Self {
46            buf: vec![0.0; len.max(1)],
47            write_idx: 0,
48            lfo,
49            sin_inc: 0.0,
50            cos_inc: 1.0,
51            cached_rate: -1.0,
52            sample_rate,
53        }
54    }
55
56    /// Process one sample.
57    ///
58    /// Returns `input` unchanged when `mix < 1e-4` (fast bypass).
59    #[allow(clippy::cast_precision_loss)]
60    pub fn process(&mut self, input: f32, params: &ChorusParams) -> f32 {
61        if params.mix < 1e-4 {
62            return input;
63        }
64
65        debug_assert!(
66            params.depth_ms <= CENTER_DELAY_MS,
67            "depth_ms ({}) exceeds center delay ({}ms) — read offset would wrap to zero-latency tap",
68            params.depth_ms,
69            CENTER_DELAY_MS
70        );
71
72        // Recompute rotation coefficients only when rate changes.
73        // Intentional exact-bits comparison: cached_rate stores the f32 we last used.
74        #[allow(clippy::float_cmp)]
75        if params.rate != self.cached_rate {
76            let phase_inc = TAU * params.rate / self.sample_rate;
77            (self.sin_inc, self.cos_inc) = crate::math::sin_cos(phase_inc);
78            self.cached_rate = params.rate;
79        }
80
81        let buf_len = self.buf.len();
82        let buf_len_f = buf_len as f32;
83        self.buf[self.write_idx] = input;
84
85        let center_samples = CENTER_DELAY_MS / 1000.0 * self.sample_rate;
86        let depth_samples = params.depth_ms.clamp(0.0, CENTER_DELAY_MS) / 1000.0 * self.sample_rate;
87        let write_idx_f = self.write_idx as f32;
88
89        let mut tap_sum = 0.0_f32;
90        for (s, c) in &mut self.lfo {
91            let offset = (center_samples + depth_samples * *s).clamp(0.0, buf_len_f - 1.001); // -1.001: keep ceil_idx in-bounds after floor+1
92
93            // Rotate LFO state — no sin() call.
94            let new_s = *s * self.cos_inc + *c * self.sin_inc;
95            let new_c = *c * self.cos_inc - *s * self.sin_inc;
96            *s = new_s;
97            *c = new_c;
98
99            // Fractional read position (backwards from write head).
100            // raw is in [buf_len_f - (buf_len_f - 1.0), buf_len_f + write_idx_f]
101            // i.e. always in (0, 2*buf_len_f), so one conditional subtract suffices.
102            let raw = write_idx_f + buf_len_f - offset;
103            let read_pos = if raw >= buf_len_f {
104                raw - buf_len_f
105            } else {
106                raw
107            };
108
109            #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
110            let floor_idx = read_pos.floor() as usize % buf_len;
111            let ceil_idx = (floor_idx + 1) % buf_len;
112            let frac = read_pos.fract();
113            tap_sum += self.buf[floor_idx] * (1.0 - frac) + self.buf[ceil_idx] * frac;
114        }
115
116        self.write_idx = (self.write_idx + 1) % buf_len;
117
118        #[allow(clippy::cast_precision_loss)]
119        let wet = tap_sum / NUM_TAPS as f32;
120        input * (1.0 - params.mix) + wet * params.mix
121    }
122}

124#[cfg(test)]
125mod tests {
126    use super::*;
127
128    fn params(rate: f32, depth_ms: f32, mix: f32) -> ChorusParams {
129        ChorusParams {
130            rate,
131            depth_ms,
132            mix,
133        }
134    }
135
136    #[test]
137    #[allow(clippy::float_cmp)]
138    fn bypass_at_zero_mix() {
139        let mut c = Chorus::new(44100.0);
140        let p = params(0.5, 3.0, 0.0);
141        for &x in &[-1.0_f32, -0.5, 0.0, 0.5, 1.0] {
142            assert_eq!(c.process(x, &p), x, "zero-mix bypass failed for {x}");
143        }
144    }
145
146    #[test]
147    fn output_finite_at_boundary_params() {
148        let mut c = Chorus::new(44100.0);
149        for &rate in &[0.1_f32, 2.5, 5.0] {
150            for &depth_ms in &[0.0_f32, 5.0, 10.0] {
151                for &mix in &[0.0_f32, 0.5, 1.0] {
152                    let p = params(rate, depth_ms, mix);
153                    for &x in &[-1.0_f32, 0.0, 1.0] {
154                        let out = c.process(x, &p);
155                        assert!(
156                            out.is_finite(),
157                            "NaN/Inf: rate={rate}, depth={depth_ms}, mix={mix}, in={x}, out={out}"
158                        );
159                    }
160                }
161            }
162        }
163    }
164
165    #[test]
166    fn modulation_active_after_warmup() {
167        // After filling the delay line, output with mix=1.0 and depth>0 must be non-zero.
168        let mut c = Chorus::new(44100.0);
169        let p = params(1.0, 5.0, 1.0);
170        // Warm up: more than MAX_CHORUS_DELAY_MS worth of samples (50ms @ 44100 = 2205 samples).
171        for _ in 0..3000 {
172            c.process(1.0, &p);
173        }
174        let out = c.process(1.0, &p);
175        assert!(
176            out.abs() > 0.01,
177            "chorus output should be non-zero after warmup: {out}"
178        );
179    }
180
181    #[test]
182    fn dc_input_steady_state_near_unity() {
183        // With depth=0 all three taps read from the same center-delay position.
184        // After the buffer fills with 1.0, tap_sum/3 = 1.0, so wet = 1.0.
185        let mut c = Chorus::new(44100.0);
186        let p = params(0.1, 0.0, 1.0); // depth=0 → no modulation
187        for _ in 0..3000 {
188            c.process(1.0, &p);
189        }
190        let out = c.process(1.0, &p);
191        assert!(
192            (out - 1.0).abs() < 0.02,
193            "expected ~1.0 for DC+depth=0, got {out}"
194        );
195    }

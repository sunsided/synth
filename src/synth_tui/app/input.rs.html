1//! Keyboard event handler: piano note mapping and application control keys.
2//!
3//! ## Piano keyboard layout
4//!
5//! Bottom octave row (maps to octave N):
6//! ```text
7//! Key:  Z  S  X  D  C  V  G  B  H  N  J  M
8//! Note: C  C# D  D# E  F  F# G  G# A  A# B
9//! ```
10//!
11//! Upper octave row (maps to octave N+1):
12//! ```text
13//! Key:  Q  2  W  3  E  R  5  T  6  Y  7  U
14//! Note: C  C# D  D# E  F  F# G  G# A  A# B
15//! ```
16//!
17//! MIDI note = 12 × (octave + 1) + semitone
18
19use crate::app::state::AppState;
20use crossterm::event::{KeyCode, KeyEvent, KeyEventKind, KeyModifiers};
21use synthie::params::MidiNote;
22
23/// Map a key code to a semitone offset (0 = C of the bottom row, 12 = C of
24/// the upper row).  Returns `None` for non-note keys.
25fn key_to_semitone(code: KeyCode) -> Option<u8> {
26    match code {
27        // Bottom row (octave N)
28        KeyCode::Char('z') => Some(0),  // C
29        KeyCode::Char('s') => Some(1),  // C#
30        KeyCode::Char('x') => Some(2),  // D
31        KeyCode::Char('d') => Some(3),  // D#
32        KeyCode::Char('c') => Some(4),  // E
33        KeyCode::Char('v') => Some(5),  // F
34        KeyCode::Char('g') => Some(6),  // F#
35        KeyCode::Char('b') => Some(7),  // G
36        KeyCode::Char('h') => Some(8),  // G#
37        KeyCode::Char('n') => Some(9),  // A
38        KeyCode::Char('j') => Some(10), // A#
39        KeyCode::Char('m') => Some(11), // B
40        // Upper row (octave N+1)
41        KeyCode::Char('q') => Some(12), // C
42        KeyCode::Char('2') => Some(13), // C#
43        KeyCode::Char('w') => Some(14), // D
44        KeyCode::Char('3') => Some(15), // D#
45        KeyCode::Char('e') => Some(16), // E
46        KeyCode::Char('r') => Some(17), // F
47        KeyCode::Char('5') => Some(18), // F#
48        KeyCode::Char('t') => Some(19), // G
49        KeyCode::Char('6') => Some(20), // G#
50        KeyCode::Char('y') => Some(21), // A
51        KeyCode::Char('7') => Some(22), // A#
52        KeyCode::Char('u') => Some(23), // B
53        _ => None,
54    }
55}
56
57/// Convert octave + semitone offset to a MIDI note number, clamped to 0..=127.
58fn make_midi(octave: i8, semitone: u8) -> MidiNote {
59    let raw = 12_i32 * (i32::from(octave) + 1) + i32::from(semitone);
60    MidiNote(u8::try_from(raw.clamp(0, 127)).expect("clamped to 0..=127"))
61}

63/// Process a single key event.  Returns `true` if the application should quit.
64pub fn handle_key(key: KeyEvent, state: &mut AppState) -> bool {
65    // Ctrl+C / Ctrl+Q → quit
66    if key.modifiers.contains(KeyModifiers::CONTROL) {
67        match key.code {
68            KeyCode::Char('c' | 'q') => return true,
69            KeyCode::Char('s') => {
70                // Ctrl+S: save current params as user patch (quick save)
71                let name = format!("User {}", state.patches.len() + 1);
72                state.save_current_as_preset(&name);
73                return false;
74            }
75            _ => return false,
76        }
77    }
78
79    // Key-release events: send note-off for piano keys.
80    if key.kind == KeyEventKind::Release {
81        if let Some(semi) = key_to_semitone(key.code) {
82            let midi = make_midi(state.octave, semi);
83            state.note_off(midi);
84        }
85        return false;
86    }
87
88    // From here on we only handle Press (and Repeat for held keys – repeat
89    // does NOT re-trigger note-on to avoid LFSR/envelope resets).
90    if key.kind == KeyEventKind::Repeat {
91        return false;
92    }
93
94    // Piano note keys (Press only)
95    if let Some(semi) = key_to_semitone(key.code) {
96        let midi = make_midi(state.octave, semi);
97        state.note_on(midi);
98        return false;
99    }
100
101    // Application control keys
102    match key.code {
103        KeyCode::F(12) => return true,
104
105        KeyCode::F(1) => {
106            state.show_help = !state.show_help;
107        }
108
109        KeyCode::Esc => {
110            state.panic_all_notes();
111        }
112
113        KeyCode::Tab => {
114            if key.modifiers.contains(KeyModifiers::SHIFT) {
115                state.prev_section();
116            } else {
117                state.next_section();
118            }
119        }
120
121        KeyCode::Left => state.prev_param(),
122        KeyCode::Right => state.next_param(),
123
124        KeyCode::Up => state.adjust_param(1.0),
125        KeyCode::Down => state.adjust_param(-1.0),
126
127        KeyCode::Enter if state.selected_section == crate::app::state::Section::Presets => {
128            let idx = state.selected_preset;
129            state.load_preset(idx);
130        }
131
132        KeyCode::Char('+' | '=') => state.volume_up(),
133        KeyCode::Char('-' | '_') => state.volume_down(),
134
135        KeyCode::Char('[' | ',') => state.octave_down(),
136        KeyCode::Char(']' | '.') => state.octave_up(),
137
138        // Quick-select presets 1–8 (number keys, only in Presets section)
139        KeyCode::Char(ch @ '1'..='8') => {
140            let idx = (ch as u8 - b'1') as usize;
141            if state.selected_section == crate::app::state::Section::Presets {
142                state.load_preset(idx);
143            }
144        }
145
146        _ => {}
147    }
148
149    false
150}

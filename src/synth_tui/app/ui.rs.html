15/// Primary foreground colour (active text).
16const FG: Color = Color::Cyan;
17/// Dimmed foreground colour (inactive labels, separators).
18const FG_DIM: Color = Color::DarkGray;
19/// Highlight colour for selected items and titles.
20const FG_HIGHLIGHT: Color = Color::Yellow;
21/// Colour used for parameter values.
22const FG_VALUE: Color = Color::Green;
23/// Background colour.
24const BG: Color = Color::Black;
25/// Border colour for the currently active panel.
26const BORDER_ACTIVE: Color = Color::Cyan;
27/// Border colour for inactive panels.
28const BORDER_INACTIVE: Color = Color::DarkGray;

30/// Top-level draw entry point.  Called once per UI frame.
31///
32/// Routes to the help overlay when `state.show_help` is set; otherwise renders
33/// the full synth UI.
34pub fn draw(frame: &mut Frame, state: &AppState, scope_data: &[(f64, f64)]) {
35    let area = frame.area();
36
37    if state.show_help {
38        draw_help(frame, area);
39        return;
40    }
41
42    // Outer layout: top body + bottom status bar
43    let outer = Layout::default()
44        .direction(Direction::Vertical)
45        .constraints([Constraint::Min(0), Constraint::Length(3)])
46        .split(area);
47
48    let body = outer[0];
49    let status_bar = outer[1];
50
51    // Body: left column (controls) + right column (scope + presets)
52    let columns = Layout::default()
53        .direction(Direction::Horizontal)
54        .constraints([Constraint::Length(26), Constraint::Min(0)])
55        .split(body);
56
57    let left = columns[0];
58    let right = columns[1];
59
60    // Left column: title + five param sections, plus a bottom spacer
61    let left_sections = Layout::default()
62        .direction(Direction::Vertical)
63        .constraints([
64            Constraint::Length(3),  // title
65            Constraint::Length(11), // OSC (9 rows + 2 border lines)
66            Constraint::Length(8),  // ENV
67            Constraint::Length(6),  // FILTER
68            Constraint::Length(5),  // LFO
69            Constraint::Length(5),  // FX
70            Constraint::Min(8),     // MOD (scrollable; 8 rows visible minimum)
71        ])
72        .split(left);
73
74    draw_title(frame, left_sections[0], state);
75    draw_section(frame, left_sections[1], state, Section::Osc);
76    draw_section(frame, left_sections[2], state, Section::Env);
77    draw_section(frame, left_sections[3], state, Section::Filter);
78    draw_section(frame, left_sections[4], state, Section::Lfo);
79    draw_section(frame, left_sections[5], state, Section::Fx);
80    draw_section(frame, left_sections[6], state, Section::Mod);
81
82    // Right column: waveform scope (top two-thirds) + presets (bottom third)
83    let right_sections = Layout::default()
84        .direction(Direction::Vertical)
85        .constraints([Constraint::Ratio(2, 3), Constraint::Ratio(1, 3)])
86        .split(right);
87
88    draw_scope(frame, right_sections[0], scope_data);
89    draw_presets(frame, right_sections[1], state);
90
91    draw_status(frame, status_bar, state);
92}

94/// Render the title bar showing the synth name and current patch name.
95fn draw_title(frame: &mut Frame, area: Rect, state: &AppState) {
96    let title = Paragraph::new(Line::from(vec![
97        Span::styled(
98            " SID SYNTH ",
99            Style::default()
100                .fg(FG_HIGHLIGHT)
101                .add_modifier(Modifier::BOLD),
102        ),
103        Span::styled(
104            format!(" {}", state.current_patch_name),
105            Style::default().fg(FG_DIM),
106        ),
107    ]))
108    .block(
109        Block::default()
110            .borders(Borders::BOTTOM)
111            .border_style(Style::default().fg(BORDER_INACTIVE)),
112    );
113    frame.render_widget(title, area);
114}

116/// Render a single parameter section panel.
117///
118/// When the section is active, all parameters are listed with the selected one
119/// highlighted.  When inactive, a compact single-line summary is shown instead.
120fn draw_section(frame: &mut Frame, area: Rect, state: &AppState, section: Section) {
121    let is_active = state.selected_section == section;
122    let border_style = Style::default().fg(if is_active {
123        BORDER_ACTIVE
124    } else {
125        BORDER_INACTIVE
126    });
127    let title_style = if is_active {
128        Style::default()
129            .fg(FG_HIGHLIGHT)
130            .add_modifier(Modifier::BOLD)
131    } else {
132        Style::default().fg(FG_DIM)
133    };
134
135    let block = Block::default()
136        .title(Span::styled(format!(" {} ", section.name()), title_style))
137        .borders(Borders::ALL)
138        .border_style(border_style);
139
140    let inner = block.inner(area);
141    frame.render_widget(block, area);
142
143    let params = state.section_params();
144    if section == Section::Presets || section != state.selected_section {
145        // Non-active section: compact single-line overview (up to 4 params)
146        if params.is_empty() {
147            return;
148        }
149        let max = params.len().min(4);
150        let items: Vec<Span> = params[..max]
151            .iter()
152            .enumerate()
153            .flat_map(|(i, (label, val))| {
154                let sep = if i > 0 {
155                    vec![Span::styled("  ", Style::default())]
156                } else {
157                    vec![]
158                };
159                let mut spans = sep;
160                if !label.is_empty() {
161                    spans.push(Span::styled(
162                        format!("{label}:"),
163                        Style::default().fg(FG_DIM),
164                    ));
165                }
166                spans.push(Span::styled(val.clone(), Style::default().fg(FG_VALUE)));
167                spans
168            })
169            .collect();
170        let para = Paragraph::new(Line::from(items));
171        frame.render_widget(para, inner);
172        return;
173    }
174
175    // Active section: full parameter list with selection cursor
176    let items: Vec<ListItem> = params
177        .iter()
178        .enumerate()
179        .map(|(i, (label, val))| {
180            let selected = i == state.selected_param;
181            let label_span = Span::styled(
182                format!("{:<7}", if label.is_empty() { " " } else { label }),
183                Style::default().fg(if selected { FG_HIGHLIGHT } else { FG_DIM }),
184            );
185            let arrow = if selected {
186                Span::styled("▶ ", Style::default().fg(FG_HIGHLIGHT))
187            } else {
188                Span::styled("  ", Style::default())
189            };
190            let val_span = Span::styled(
191                val.clone(),
192                Style::default()
193                    .fg(if selected { FG_HIGHLIGHT } else { FG_VALUE })
194                    .add_modifier(if selected {
195                        Modifier::BOLD
196                    } else {
197                        Modifier::empty()
198                    }),
199            );
200            ListItem::new(Line::from(vec![arrow, label_span, val_span]))
201        })
202        .collect();
203
204    let list = List::new(items);
205    let mut list_state = ListState::default().with_selected(Some(state.selected_param));
206    frame.render_stateful_widget(list, inner, &mut list_state);
207}

209/// Render the waveform scope chart.
210#[allow(clippy::cast_precision_loss)] // sample count fits well within f64 mantissa in practice
211fn draw_scope(frame: &mut Frame, area: Rect, data: &[(f64, f64)]) {
212    let block = Block::default()
213        .title(Span::styled(" WAVEFORM ", Style::default().fg(FG_DIM)))
214        .borders(Borders::ALL)
215        .border_style(Style::default().fg(BORDER_INACTIVE));
216
217    if data.is_empty() {
218        frame.render_widget(
219            Paragraph::new("no signal")
220                .style(Style::default().fg(FG_DIM))
221                .alignment(Alignment::Center)
222                .block(block),
223            area,
224        );
225        return;
226    }
227
228    let x_max = data.len() as f64;
229
230    let datasets = vec![
231        Dataset::default()
232            .name("")
233            .marker(symbols::Marker::Braille)
234            .graph_type(GraphType::Line)
235            .style(Style::default().fg(FG))
236            .data(data),
237    ];
238
239    let chart = Chart::new(datasets)
240        .block(block)
241        .x_axis(
242            Axis::default()
243                .style(Style::default().fg(FG_DIM))
244                .bounds([0.0, x_max]),
245        )
246        .y_axis(
247            Axis::default()
248                .style(Style::default().fg(FG_DIM))
249                .bounds([-1.0, 1.0])
250                .labels([Span::raw("-1"), Span::raw(" 0"), Span::raw("+1")]),
251        );
252
253    frame.render_widget(chart, area);
254}

256/// Render the preset list panel.
257fn draw_presets(frame: &mut Frame, area: Rect, state: &AppState) {
258    let is_active = state.selected_section == Section::Presets;
259    let border_style = Style::default().fg(if is_active {
260        BORDER_ACTIVE
261    } else {
262        BORDER_INACTIVE
263    });
264    let title_style = if is_active {
265        Style::default()
266            .fg(FG_HIGHLIGHT)
267            .add_modifier(Modifier::BOLD)
268    } else {
269        Style::default().fg(FG_DIM)
270    };
271
272    let block = Block::default()
273        .title(Span::styled(" PRESETS ", title_style))
274        .borders(Borders::ALL)
275        .border_style(border_style);
276
277    let inner = block.inner(area);
278    frame.render_widget(block, area);
279
280    let items: Vec<ListItem> = state
281        .patches
282        .iter()
283        .enumerate()
284        .map(|(i, patch)| {
285            let selected = i == state.selected_preset;
286            let style = if selected && is_active {
287                Style::default()
288                    .fg(FG_HIGHLIGHT)
289                    .add_modifier(Modifier::BOLD | Modifier::REVERSED)
290            } else if selected {
291                Style::default().fg(FG_VALUE).add_modifier(Modifier::BOLD)
292            } else {
293                Style::default().fg(FG_DIM)
294            };
295            ListItem::new(Line::from(Span::styled(
296                format!(" {:>2}. {}", i + 1, patch.name),
297                style,
298            )))
299        })
300        .collect();
301
302    let mut list_state = ListState::default();
303    list_state.select(Some(state.selected_preset));
304
305    frame.render_stateful_widget(List::new(items), inner, &mut list_state);
306}

308/// Render the status bar with note, octave, volume, and key binding hints.
309fn draw_status(frame: &mut Frame, area: Rect, state: &AppState) {
310    let note_str = match state.active_note {
311        Some(midi) => {
312            let names = [
313                "C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B",
314            ];
315            let raw = midi.as_u8();
316            let name = names[(raw % 12) as usize];
317            let oct = i32::from(raw / 12) - 1;
318            format!("{name}{oct}({raw})")
319        }
320        None => "---".to_string(),
321    };
322
323    let msg = if state.status_msg.is_empty() {
324        String::new()
325    } else {
326        format!("  {}  ", state.status_msg)
327    };
328
329    let bar = Paragraph::new(Line::from(vec![
330        Span::styled(" Note:", Style::default().fg(FG_DIM)),
331        Span::styled(format!("{note_str} "), Style::default().fg(FG_VALUE)),
332        Span::styled("|", Style::default().fg(FG_DIM)),
333        Span::styled(
334            format!(" Oct:{} ", state.octave),
335            Style::default().fg(FG_VALUE),
336        ),
337        Span::styled("|", Style::default().fg(FG_DIM)),
338        Span::styled(
339            format!(" Vol:{:.0}% ", state.params.global.volume * 100.0),
340            Style::default().fg(FG_VALUE),
341        ),
342        Span::styled("|", Style::default().fg(FG_DIM)),
343        Span::styled(msg, Style::default().fg(FG_HIGHLIGHT)),
344        Span::styled(
345            " Tab:section  ←→:param  ↑↓:value  []:octave  F1:help  Esc:panic  ^C:quit",
346            Style::default().fg(FG_DIM),
347        ),
348    ]))
349    .block(
350        Block::default()
351            .borders(Borders::TOP)
352            .border_style(Style::default().fg(BORDER_INACTIVE)),
353    );
354
355    frame.render_widget(bar, area);
356}

461/// Return a centred `Rect` that is `percent_x`% wide and `percent_y`% tall
462/// relative to `area`.
463fn centred_rect(percent_x: u16, percent_y: u16, area: Rect) -> Rect {
464    let layout = Layout::default()
465        .direction(Direction::Vertical)
466        .constraints([
467            Constraint::Percentage((100 - percent_y) / 2),
468            Constraint::Percentage(percent_y),
469            Constraint::Percentage((100 - percent_y) / 2),
470        ])
471        .split(area);
472
473    Layout::default()
474        .direction(Direction::Horizontal)
475        .constraints([
476            Constraint::Percentage((100 - percent_x) / 2),
477            Constraint::Percentage(percent_x),
478            Constraint::Percentage((100 - percent_x) / 2),
479        ])
480        .split(layout[1])[1]
481}

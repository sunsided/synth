1//! Rolling waveform scope buffer.
2//!
3//! The audio thread sends batches of decimated `f32` samples over a bounded
4//! channel.  The UI calls [`ScopeBuf::update`] each frame to drain any pending
5//! batches, then reads [`ScopeBuf::as_chart_data`] to feed the ratatui chart.
6
7use crossbeam_channel::Receiver;
8use std::collections::VecDeque;
9
10/// Rolling waveform scope buffer.
11///
12/// The audio thread sends batches of decimated samples via a bounded channel.
13/// The UI calls `update()` each frame to drain new batches, then reads
14/// `samples()` or `as_chart_data()` for display.
15pub struct ScopeBuf {
16    /// Channel receiver for incoming sample batches from the audio thread.
17    rx: Receiver<Vec<f32>>,
18    /// Ring buffer holding the most recent `capacity` samples.
19    buf: VecDeque<f32>,
20    /// Maximum number of samples retained in the ring buffer.
21    capacity: usize,
22}
23
24impl ScopeBuf {
25    /// Create a new scope buffer with the given channel receiver and sample capacity.
26    pub fn new(rx: Receiver<Vec<f32>>, capacity: usize) -> Self {
27        Self {
28            rx,
29            buf: VecDeque::with_capacity(capacity),
30            capacity,
31        }
32    }
33
34    /// Drain all pending batches from the channel into the rolling buffer.
35    ///
36    /// Should be called once per UI frame before reading scope data.
37    pub fn update(&mut self) {
38        while let Ok(batch) = self.rx.try_recv() {
39            for s in batch {
40                if self.buf.len() >= self.capacity {
41                    self.buf.pop_front();
42                }
43                self.buf.push_back(s);
44            }
45        }
46    }
47
48    /// Iterate over the current waveform data (oldest → newest).
49    #[allow(dead_code)]
50    pub fn samples(&self) -> impl Iterator<Item = f32> + '_ {
51        self.buf.iter().copied()
52    }
53
54    /// Return waveform data as `(x, y)` pairs suitable for a ratatui `Dataset`.
55    #[allow(clippy::cast_precision_loss)] // scope index fits within f64 mantissa for display sizes
56    pub fn as_chart_data(&self) -> Vec<(f64, f64)> {
57        self.buf
58            .iter()
59            .enumerate()
60            .map(|(i, &s)| (i as f64, f64::from(s)))
61            .collect()
62    }
63
64    /// Number of samples currently in the buffer.
65    #[allow(dead_code)]
66    pub fn len(&self) -> usize {
67        self.buf.len()
68    }
69
70    /// Returns `true` if the buffer contains no samples.
71    #[allow(dead_code)]
72    pub fn is_empty(&self) -> bool {
73        self.buf.is_empty()
74    }
75}
